"""G4 (BRIEF_46 S.4, step 7): the numbers the packet's "For the paper" section quotes, read from the five committed
artifacts by program -- every registered sentence's slots, every moved verdict. Rounded to 2 decimals for display;
the artifacts hold the exact values. att_engine is primary (A27(a)); att_ours beside.

Usage: python -P for_the_paper.py REPO
"""
import json
import sys
from pathlib import Path

DATA = Path(sys.argv[1]) / "docs" / "data"


def load(name):
    return json.loads((DATA / name).read_text(encoding="utf-8"))


def ci(block):
    return f"{block['mean_difference']:+.2f} [{block['ci95_low']:+.2f}, {block['ci95_high']:+.2f}]"


def both(entry, key="by_definition"):
    e, o = entry[key]["att_engine"], entry[key]["att_ours"]
    out = f"att_engine {ci(e)}"
    if "outcome" in e:
        out += f" {e['outcome']}"
    out += f" | att_ours {ci(o)}"
    if "outcome" in o:
        out += f" {o['outcome']}"
    return out

d = load("p9_1_exposure_parity.json")
grid = d["exposure_table"]["grid4x4"]["arms"]
print("  exposure (grid, row visits over 40,000 steps):", {k: v.get("row_visits", v) if isinstance(v, dict) else v
                                                         for k, v in list(grid.items())[:11]})
for kind in ("original", "matched"):
    for e in d["sentence_contrasts"][kind]:
        print(f"  sentence, {kind:8s}: {e['left_arm']} - {e['right_arm']} on {e['tier']}: {both(e)}")
r = d["rescore"]
print(f"  rescore: {len(d['counted_statement_ids'])} statement ids; verdicts per definition {r['n_statements']}; "
      f"moved {r['n_moved']}")
moved = [e for e in r["statements"] if e["moved"]]
print("  moved verdicts:", [(e["id"], e["tier"], e["definition"], e["original_verdict"], e.get("verdict")) for e in moved])
changed = [e for e in r["statements"] if e["reading_changed"] and not e["moved"]]

          f"{e['original_reading'][:60]!r} -> {e['reading'][:60]!r}")
